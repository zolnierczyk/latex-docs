\newcommand{\MakeDailyPageWithWeeklyCheck}[3]{%
    \pgfcalendar{checkday}{#1-#2-#3}{#1-#2-#3}{%
        \MakeDailyPage{#1}{#2}{#3}{\pgfcalendarcurrentweekday}%
        \ifnum\pgfcalendarcurrentweekday=6\relax
            \MakeWeeklySummaryPage{}{}{}%
        \fi
    }%
}
