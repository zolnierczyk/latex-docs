% --- PREAMBUŁA ---
% Tutaj definiujemy podstawowe ustawienia dokumentu i wczytujemy potrzebne pakiety.

\documentclass[a4paper, 12pt]{article} % Definiuje klasę dokumentu: artykuł, rozmiar papieru A4, rozmiar czcionki 12pt.

% --- PAKIETY ---
\usepackage[utf8]{inputenc} % Umożliwia wprowadzanie polskich znaków (ą, ę, ć, etc.) bezpośrednio z klawiatury.
\usepackage[T1]{fontenc}      % Zapewnia poprawne kodowanie czcionek, co jest kluczowe dla polskich znaków.
\usepackage{polski}           % Włącza polskie zasady składu tekstu (np. dzielenie wyrazów).
\usepackage[margin=2cm]{geometry} % Ustawia marginesy strony na 2cm z każdej strony, aby maksymalnie wykorzystać przestrzeń.
\usepackage{multicol}         % Umożliwia tworzenie układu wielokolumnowego.
\usepackage{amssymb}          % Dostarcza symbole matematyczne, w tym symbol pustego kwadratu (\square).
\usepackage{enumitem}         % Pozwala na zaawansowaną personalizację list (np. usunięcie wcięć).
\usepackage{titlesec}         % Umożliwia modyfikację wyglądu tytułów sekcji.
\usepackage{helvet}

% --- USTAWIENIA WYGLĄDU ---
\pagestyle{empty} % Usuwa numery stron, aby arkusz był czysty i gotowy do druku.

% Modyfikacja wyglądu tytułów sekcji, aby były większe i wyśrodkowane.
\titleformat{\section}
  {\normalfont\Large\bfseries\centering} % Formatowanie: normalna czcionka, duży rozmiar, pogrubiona, wyśrodkowana.
  {} % Miejsce na numer sekcji (nie będzie wyświetlany, bo użyjemy \section*).
  {0em} % Odstęp między numerem a tytułem.
  {} % Kod przed tytułem.

% Definicja pomocniczych poleceń do wstawiania checkboxów.
% To sprawia, że kod jest czystszy i łatwiejszy w zarządzaniu.
% \hfil automatycznie przesuwa kwadraty do prawego marginesu.

% Polecenie wstawiające 7 kwadratów dla zadań codziennych.
\newcommand{\dailyboxes}{%
    \leavevmode\unskip\nobreak%
    {\Large$\square$}%
    \hspace{1mm}{\Large$\square$}%
    \hspace{1mm}{\Large$\square$}%
    \hspace{1mm}{\Large$\square$}%
    \hspace{1mm}{\Large$\square$}%
    \hspace{1mm}{\Large$\square$}%
    \hspace{1mm}{\Large$\square$}%
    \hspace{1ex}
}

\newcommand{\dailyboxesDayNames}{%
    \leavevmode\unskip\nobreak%
    P
    \hspace{1.5mm}W
    \hspace{1.5mm}Ś
    \hspace{1.5mm}C
    \hspace{1.5mm}P
    \hspace{1.5mm}S
    \hspace{1.5mm}N
    \hspace{1ex}
}

% Polecenie wstawiające 3 kwadratów dla zadań dwa razy w tygodniu.
\newcommand{\midweekboxes}{%
    \leavevmode\unskip\nobreak%
    {\Large$\square$}%
    \hspace{1mm}{\Large$\square$}%
    \hspace{1mm}{\Large$\square$}%
    \hspace{1ex}
}

% Polecenie wstawiające 1 kwadrat dla zadań tygodniowych.
\newcommand{\weeklybox}{%
    \leavevmode\unskip\nobreak%
    {\Large$\square$}%
    \hspace{1ex}
}

% Ustawienia dla list, aby nie miały wcięć i standardowych punktorów.
% Dzięki temu lista wygląda jak schludna checklista.
\setlist[itemize]{
  label=, % Usuwa domyślne punktory (np. kropki).
  leftmargin=*, % Usuwa wcięcie z lewej strony.
  itemsep=2ex % Zwiększa odstęp między elementami listy dla lepszej czytelności.
}
